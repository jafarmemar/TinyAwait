cmake_minimum_required(VERSION 3.20)
project(TinyAwait VERSION 1.0.1 LANGUAGES CXX)

add_library(TinyAwait INTERFACE)
add_library(TinyAwait::TinyAwait ALIAS TinyAwait)
target_include_directories(TinyAwait INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(TinyAwait INTERFACE cxx_std_20)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(_tinyawait_top_level ON)
else()
  set(_tinyawait_top_level OFF)
endif()

option(TINYAWAIT_BUILD_TESTS "Build TinyAwait tests" ${_tinyawait_top_level})
option(TINYAWAIT_BUILD_BENCHMARKS "Build TinyAwait benchmarks" ${_tinyawait_top_level})
unset(_tinyawait_top_level)

if(TINYAWAIT_BUILD_TESTS)
  enable_testing()

  function(add_tinyawait_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE TinyAwait::TinyAwait)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endfunction()

  add_tinyawait_test(test_basic tests/test_basic.cpp)
  add_tinyawait_test(test_wraparound tests/test_wraparound.cpp)
  add_tinyawait_test(test_capacity tests/test_capacity.cpp)
  add_tinyawait_test(test_no_heap tests/test_no_heap.cpp)
  add_tinyawait_test(test_stress tests/test_stress.cpp)
  add_tinyawait_test(test_frame_size tests/test_frame_size.cpp)
  add_tinyawait_test(test_nested tests/test_nested.cpp)
  add_tinyawait_test(test_max_delay tests/test_max_delay.cpp)
  add_tinyawait_test(test_default_capacity tests/test_default_capacity.cpp)
  add_tinyawait_test(test_delay_validation tests/test_delay_validation.cpp)
  add_tinyawait_test(test_randomized tests/test_randomized.cpp)
  add_tinyawait_test(test_multitu tests/test_multitu_main.cpp tests/test_multitu_peer.cpp)
  add_tinyawait_test(test_next_deadline tests/test_next_deadline.cpp)
  add_tinyawait_test(test_freelist tests/test_freelist.cpp)
endif()

if(TINYAWAIT_BUILD_BENCHMARKS)
  add_executable(benchmark_tinyawait benchmarks/benchmark.cpp)
  target_link_libraries(benchmark_tinyawait PRIVATE TinyAwait::TinyAwait)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(benchmark_tinyawait PRIVATE -O2 -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
  endif()

  add_executable(benchmark_tinyawait_matrix benchmarks/benchmark_matrix.cpp)
  target_link_libraries(benchmark_tinyawait_matrix PRIVATE TinyAwait::TinyAwait)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(benchmark_tinyawait_matrix PRIVATE -O2 -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
  endif()
endif()
